cmake_minimum_required(VERSION 3.16)
project(shield_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
option(SHIELD_ENABLE_EXPERIMENTAL "Allow uncertified modes for offline certification" OFF)
add_library(shield SHARED api.cpp weights.cpp frontend.cpp kernels.cpp precision.cpp sequence.cpp)
if(SHIELD_ENABLE_EXPERIMENTAL)
  target_compile_definitions(shield PRIVATE SHIELD_ENABLE_EXPERIMENTAL=1)
endif()
target_include_directories(shield PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
# Baseline ISA intentionally portable. Individual measured kernels use function
# target attributes and runtime feature checks; never compile with -march=native.
target_compile_options(shield PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Wpedantic)
target_link_libraries(shield PRIVATE Threads::Threads)
set_target_properties(shield PROPERTIES CXX_VISIBILITY_PRESET default)
install(TARGETS shield LIBRARY DESTINATION lib)
install(FILES include/shield.h DESTINATION include)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../bench/native_bench.cpp)
  add_executable(shield_bench ../bench/native_bench.cpp)
  target_compile_options(shield_bench PRIVATE -O2)
  target_link_libraries(shield_bench PRIVATE shield)
endif()
